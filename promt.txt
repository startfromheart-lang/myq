# 麻友圈 - 麻将社交匹配平台全栈应用生成提示词

## 项目概述
请生成一个名为“麻友圈”的H5全栈Web应用，帮助麻将爱好者找到合适的牌友并预约麻将馆。系统包含C端（用户端）和B端（商家端）两大模块，使用Next.js 14 (App Router)、TypeScript、Prisma ORM、SQLite数据库、Tailwind CSS，并采用移动优先设计。

## 技术栈要求
- **前端框架**: Next.js 14 (App Router) + React Server Components
- **样式**: Tailwind CSS + shadcn/ui 组件库
- **后端**: Next.js API Routes (App Router)
- **数据库**: SQLite + Prisma ORM
- **认证**: NextAuth.js (Credentials Provider + 手机号验证码登录)
- **文件上传**: 使用 multer 或 formidable 存储到本地 `public/uploads/`
- **实时通信**: Socket.io (群聊功能) 或简单轮询实现
- **地图定位**: 使用 @react-google-maps/api 或 Leaflet (模拟坐标展示)
- **支付**: 模拟支付接口 (无需真实对接)
- **人脸识别**: 模拟实现 (调用第三方服务mock或本地图片对比模拟)

## 数据库模型设计 (Prisma Schema)

```prisma
// 请在生成的代码中提供完整的schema.prisma文件
model User {
  id               String    @id @default(cuid())
  phone            String    @unique
  isVerified       Boolean   @default(false)  // 实名认证状态
  realName         String?
  idCard           String?   // 身份证号
  facePhoto        String?   // 扫脸照片路径
  avatar           String?   // 头像路径
  selfEvaluation   String?   // 自我评价
  preferredMahjong String[]  // 希望的麻将种类 ["敲麻","川麻","红中麻将"]
  job              String?   // 工作
  gender           String?   // 性别
  city             String?   // 城市
  skillScore       Float     @default(5.0)  // 麻将技术分 (1-10)
  integrityScore   Float     @default(5.0)  // 诚信分
  personalityTags  String[]  // 性格标签
  createdAt        DateTime  @default(now())
  updatedAt        DateTime  @updatedAt

  // 关系
  matchGroups      MatchGroupParticipant[]
  reviewsGiven     Review[]   @relation("Reviewer")
  reviewsReceived  Review[]   @relation("Reviewee")
  payments         Payment[]
}

model Merchant {
  id               String    @id @default(cuid())
  inviteCode       String    @unique
  businessLicense  String    // 营业执照图片路径
  legalPersonName  String
  legalPersonId    String
  contractSigned   Boolean   @default(false)
  disclaimerAccepted Boolean @default(false)
  shopName         String
  contactPhone     String
  status           String    @default("pending") // pending, approved, rejected
  createdAt        DateTime  @default(now())
  rooms            MahjongRoom[]
  coupons          Coupon[]
  campaigns        Campaign[]
}

model MahjongRoom {
  id               String    @id @default(cuid())
  merchantId       String
  name             String
  hasToilet        Boolean
  area             Float      // 面积(m²)
  mahjongSize      String     // 麻将大小
  maxTileCount     Int        // 最大章数
  price            Float      // 台费/小时
  address          String
  latitude         Float?
  longitude        Float?
  images           String[]   // 房间图片路径列表
  status           String     @default("active")
  merchant         Merchant   @relation(fields: [merchantId], references: [id])
  orders           Order[]
}

model Order {
  id               String    @id @default(cuid())
  matchGroupId     String
  roomId           String
  totalAmount      Float
  paymentStatus    String    @default("pending") // pending, paid, refunded
  checkinTime      DateTime?
  checkinCode      String?   // 核销码
  createdAt        DateTime  @default(now())
  room             MahjongRoom @relation(fields: [roomId], references: [id])
  matchGroup       MatchGroup @relation(fields: [matchGroupId], references: [id])
  payments         Payment[]
}

model Payment {
  id               String    @id @default(cuid())
  userId           String
  orderId          String
  amount           Float
  status           String    @default("pending") // pending, success, refunded
  paymentMethod    String    // "balance", "mock_wallet"
  createdAt        DateTime  @default(now())
  user             User      @relation(fields: [userId], references: [id])
  order            Order     @relation(fields: [orderId], references: [id])
}

model MatchGroup {
  id               String    @id @default(cuid())
  status           String    @default("matching") // matching, confirmed, playing, finished, failed
  mahjongType      String    // 麻将种类
  regionRange      String    // 约麻区域范围
  scheduledTime    DateTime? // 约定时间
  duration         Int?      // 时长(小时)
  roomId           String?   // 最终确定的麻将馆
  createdAt        DateTime  @default(now())
  participants     MatchGroupParticipant[]
  chatMessages     ChatMessage[]
  order            Order?
}

model MatchGroupParticipant {
  id               String    @id @default(cuid())
  groupId          String
  userId           String
  role             String    @default("member") // member, payer
  status           String    @default("pending") // pending, confirmed, declined
  paymentConfirmed Boolean   @default(false) // 是否已支付台费
  group            MatchGroup @relation(fields: [groupId], references: [id])
  user             User      @relation(fields: [userId], references: [id])
}

model ChatMessage {
  id               String    @id @default(cuid())
  groupId          String
  senderId         String
  content          String
  createdAt        DateTime  @default(now())
  group            MatchGroup @relation(fields: [groupId], references: [id])
  sender           User      @relation(fields: [senderId], references: [id])
}

model Review {
  id               String    @id @default(cuid())
  reviewerId       String
  revieweeId       String
  matchGroupId     String
  skillScore       Int       // 1-10 麻将技术分
  integrityScore   Int       // 1-10 诚信分
  personalityTags  String[]  // 性格标签选择
  content          String?
  createdAt        DateTime  @default(now())
  reviewer         User      @relation("Reviewer", fields: [reviewerId], references: [id])
  reviewee         User      @relation("Reviewee", fields: [revieweeId], references: [id])
  matchGroup       MatchGroup @relation(fields: [matchGroupId], references: [id])
}

model ShopReview {
  id               String    @id @default(cuid())
  roomId           String
  userId           String
  environmentScore Int       // 1-5
  serviceScore     Int
  cateringScore    Int
  content          String?
  createdAt        DateTime  @default(now())
  room             MahjongRoom @relation(fields: [roomId], references: [id])
  user             User      @relation(fields: [userId], references: [id])
}

model Coupon {
  id               String    @id @default(cuid())
  merchantId       String
  title            String
  discount         Float     // 折扣或金额
  type             String    // "fixed" or "percentage"
  validUntil       DateTime
  usageLimit       Int       // 总可用次数
  usedCount        Int       @default(0)
  merchant         Merchant  @relation(fields: [merchantId], references: [id])
}

model Campaign {
  id               String    @id @default(cuid())
  merchantId       String
  title            String
  description      String
  startDate        DateTime
  endDate          DateTime
  prize            String
  merchant         Merchant  @relation(fields: [merchantId], references: [id])
}
